API Overview

The Simple Todo API is a RESTful service built with Node.js, Express, and MongoDB. It provides user authentication and todo management capabilities with JWT-based security.

Authentication

All todo endpoints require JWT authentication. Include the token in the Authorization header:

Authorization Header 
Authorization: Bearer your_jwt_token_here

Response Format

All API responses follow a consistent format:

Standard Response 
{
  "success": true/false,
  "message": "Descriptive message",
  "data": {
    // Response data
  }
}

Error Handling

Error responses include appropriate HTTP status codes and descriptive messages.

POST /auth/register

Register a new user account

Request Body

Parameter Type Required Description
username string Required Unique username (3-30 characters)
email string Required Valid email address

GET /todos

Get all todos for the authenticated user with optional filtering and pagination

This endpoint requires authentication

Query Parameters

Parameter Type Required Description
completed boolean Optional Filter by completion status (true/false)
priority string Optional Filter by priority (low, medium, high)
sortBy string Optional Sort field (createdAt, title, priority). Default: createdAt
order string Optional Sort order (asc, desc). Default: desc
page number Optional Page number for pagination. Default: 1
limit number Optional Items per page. Default: 20

Example Request

cURL 
curl -X GET "http://localhost:3002/api/todos?completed=false&priority=high&page=1&limit=10" \
  -H "Authorization: Bearer eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9..."

JavaScript Examples

User Registration & Login

JavaScript 
// Register a new user
async function registerUser(userData) {
  try {
    const response = await fetch('http://localhost:3002/api/auth/register', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify(userData)
    });
    
    const data = await response.json();
    
    if (data.success) {
      // Store the token
      localStorage.setItem('token', data.data.token);
      console.log('User registered:', data.data.user);
    } else {
      console.error('Registration failed:', data.message);
    }
  } catch (error) {
    console.error('Error:', error);
  }
}

// Login user
async function loginUser(credentials) {
  try {
    const response = await fetch('http://localhost:3002/api/auth/login', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify(credentials)
    });
    
    const data = await response.json();
    
    if (data.success) {
      // Store the token
      localStorage.setItem('token', data.data.token);
      return data.data;
    } else {
      throw new Error(data.message);
    }
  } catch (error) {
    console.error('Login error:', error);
    throw error;
  }
}

// Usage
registerUser({
  username: 'johndoe',
  email: 'john@example.com',
  password: 'securepassword123'
});

loginUser({
  email: 'john@example.com',
  password: 'securepassword123'
});

Todo Management

JavaScript 
// Helper function to get auth headers
function getAuthHeaders() {
  const token = localStorage.getItem('token');
  return {
    'Content-Type': 'application/json',
    'Authorization': `Bearer ${token}`
  };
}

// Get all todos
async function getTodos(filters = {}) {
  try {
    const queryParams = new URLSearchParams(filters);
    const response = await fetch(`http://localhost:3002/api/todos?${queryParams}`, {
      headers: getAuthHeaders()
    });
    
    const data = await response.json();
    return data.success ? data.data : null;
  } catch (error) {
    console.error('Error fetching todos:', error);
  }
}

// Create new todo
async function createTodo(todoData) {
  try {
    const response = await fetch('http://localhost:3002/api/todos', {
      method: 'POST',
      headers: getAuthHeaders(),
      body: JSON.stringify(todoData)
    });
    
    const data = await response.json();
    return data.success ? data.data.todo : null;
  } catch (error) {
    console.error('Error creating todo:', error);
  }
}

// Update todo
async function updateTodo(id, updates) {
  try {
    const response = await fetch(`http://localhost:3002/api/todos/${id}`, {
      method: 'PUT',
      headers: getAuthHeaders(),
      body: JSON.stringify(updates)
    });
    
    const data = await response.json();
    return data.success ? data.data.todo : null;
  } catch (error) {
    console.error('Error updating todo:', error);
  }
}

// Toggle todo completion
async function toggleTodo(id) {
  try {
    const response = await fetch(`http://localhost:3002/api/todos/${id}/toggle`, {
      method: 'PATCH',
      headers: getAuthHeaders()
    });
    
    const data = await response.json();
    return data.success ? data.data.todo : null;
  } catch (error) {
    console.error('Error toggling todo:', error);
  }
}

// Delete todo
async function deleteTodo(id) {
  try {
    const response = await fetch(`http://localhost:3002/api/todos/${id}`, {
      method: 'DELETE',
      headers: getAuthHeaders()
    });
    
    const data = await response.json();
    return data.success;
  } catch (error) {
    console.error('Error deleting todo:', error);
    return false;
  }
}

React Hook Example

React Hook 
import { useState, useEffect } from 'react';

// Custom hook for managing todos
function useTodos() {
  const [todos, setTodos] = useState([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);

  const getAuthHeaders = () => ({
    'Content-Type': 'application/json',
    'Authorization': `Bearer ${localStorage.getItem('token')}`
  });

  const fetchTodos = async (filters = {}) => {
    setLoading(true);
    setError(null);
    
    try {
      const queryParams = new URLSearchParams(filters);
      const response = await fetch(`http://localhost:3002/api/todos?${queryParams}`, {
        headers: getAuthHeaders()
      });
      
      const data = await response.json();
      
      if (data.success) {
        setTodos(data.data.todos);
      } else {
        setError(data.message);
      }
    } catch (err) {
      setError('Failed to fetch todos');
    } finally {
      setLoading(false);
    }
  };

  const createTodo = async (todoData) => {
    try {
      const response = await fetch('http://localhost:3002/api/todos', {
        method: 'POST',
        headers: getAuthHeaders(),
        body: JSON.stringify(todoData)
      });
      
      const data = await response.json();
      
      if (data.success) {
        setTodos(prev => [data.data.todo, ...prev]);
        return data.data.todo;
      } else {
        setError(data.message);
        return null;
      }
    } catch (err) {
      setError('Failed to create todo');
      return null;
    }
  };

  const updateTodo = async (id, updates) => {
    try {
      const response = await fetch(`http://localhost:3002/api/todos/${id}`, {
        method: 'PUT',
        headers: getAuthHeaders(),
        body: JSON.stringify(updates)
      });
      
      const data = await response.json();
      
      if (data.success) {
        setTodos(prev => 
          prev.map(todo => 
            todo._id === id ? data.data.todo : todo
          )
        );
        return data.data.todo;
      } else {
        setError(data.message);
        return null;
      }
    } catch (err) {
      setError('Failed to update todo');
      return null;
    }
  };

  const deleteTodo = async (id) => {
    try {
      const response = await fetch(`http://localhost:3002/api/todos/${id}`, {
        method: 'DELETE',
        headers: getAuthHeaders()
      });
      
      const data = await response.json();
      
      if (data.success) {
        setTodos(prev => prev.filter(todo => todo._id !== id));
        return true;
      } else {
        setError(data.message);
        return false;
      }
    } catch (err) {
      setError('Failed to delete todo');
      return false;
    }
  };

  return {
    todos,
    loading,
    error,
    fetchTodos,
    createTodo,
    updateTodo,
    deleteTodo,
    setError
  };
}
